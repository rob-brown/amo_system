cmake_minimum_required(VERSION 3.13)

project(nordicpad_tests C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)

include(${CMAKE_CURRENT_SOURCE_DIR}/../cmake/nordicpad_core.cmake)

target_compile_options(nordicpad_core PRIVATE -Wall -Wextra -Werror)

enable_testing()

function(add_nordicpad_test name)
    add_executable(test_${name} test_${name}.c)
    target_include_directories(test_${name} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    target_link_libraries(test_${name} nordicpad_core)
    add_test(NAME ${name} COMMAND test_${name})
endfunction()

set(FIXTURE_DIR ${CMAKE_CURRENT_SOURCE_DIR}/fixtures)
set(GENERATED_IDENTITY ${CMAKE_CURRENT_SOURCE_DIR}/../src/generated/identity_data.c)

if(EXISTS ${FIXTURE_DIR}/init_exchanges.txt AND EXISTS ${GENERATED_IDENTITY})
    add_library(nordicpad_identity STATIC ${GENERATED_IDENTITY})
    target_link_libraries(nordicpad_identity PUBLIC nordicpad_core)
    add_nordicpad_test(init_replay)
    target_link_libraries(test_init_replay nordicpad_identity)
    target_compile_definitions(test_init_replay PRIVATE FIXTURE_DIR="${FIXTURE_DIR}")

    if(EXISTS ${FIXTURE_DIR}/nfc_v2_scan.txt AND EXISTS ${FIXTURE_DIR}/nfc_v2_dump.bin)
        add_nordicpad_test(nfc_replay)
        target_link_libraries(test_nfc_replay nordicpad_identity)
        target_compile_definitions(test_nfc_replay PRIVATE FIXTURE_DIR="${FIXTURE_DIR}")
        if(EXISTS ${FIXTURE_DIR}/nfc_v3_scan.txt AND EXISTS ${FIXTURE_DIR}/nfc_v3_dump.bin)
            target_compile_definitions(test_nfc_replay PRIVATE NFC_V3_FIXTURES)
        endif()
    else()
        message(WARNING "NFC golden test skipped: run nordicpad/scripts/extract_nfc_fixture.exs on a capture first")
    endif()
else()
    message(WARNING "Golden tests skipped: run nordicpad/scripts/extract_fixtures.exs on a capture first")
endif()
